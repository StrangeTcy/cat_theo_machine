  Source (training.py):
    class ObligationSkeletonConclusionGoal(M.Edge):
        """The last skeleton entry that carries a goal term; that goal is the
           conclusion."""                                    # docstring L244
        def _last_goal(self, skeleton):                      # L250
            if M.IdentityCompare(skeleton, M.EmptyList)() is M.truth_value:
                return M.EmptyList
            goal = ObligationSkeletonEntryGoal(M.Head(skeleton)())()  # L253
            rest = self._last_goal(M.Tail(skeleton)())()     # L254
            if M.IdentityCompare(goal, M.EmptyList)() is M.false_value:
                return goal                                  # L256
            return rest

  Empirical (real E2 record, real loader, 4 entries):
